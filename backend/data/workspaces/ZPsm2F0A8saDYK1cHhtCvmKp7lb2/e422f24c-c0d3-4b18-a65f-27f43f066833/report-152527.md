# Expense Tracker in Python with CSV Backend

## Overview
A simple expense tracker application with CSV storage, supporting add, update, and list operations.

## Key Features
- **CSV Storage**: Persistent storage of expenses in `expenses.csv`
- **CRUD Operations**: Add, update, and list expenses
- **Demo Run**: Shows complete workflow with sample data

## Code Implementation

```python
import csv
import os
from datetime import datetime

class ExpenseTracker:
    def __init__(self, csv_file="expenses.csv"):
        self.csv_file = csv_file
        self._init_csv()

    def _init_csv(self):
        if not os.path.exists(self.csv_file):
            with open(self.csv_file, 'w', newline='') as f:
                writer = csv.writer(f)
                writer.writerow(['id', 'date', 'description', 'amount', 'category'])

    def add_expense(self, description, amount, category):
        with open(self.csv_file, 'r') as f:
            reader = csv.reader(f)
            ids = [int(row[0]) for row in reader if row] or [0]
            new_id = max(ids) + 1

        with open(self.csv_file, 'a', newline='') as f:
            writer = csv.writer(f)
            writer.writerow([new_id, datetime.now().strftime('%Y-%m-%d'), description, amount, category])
        return new_id

    def update_expense(self, expense_id, description=None, amount=None, category=None):
        rows = []
        updated = False
        with open(self.csv_file, 'r') as f:
            reader = csv.reader(f)
            for row in reader:
                if row and int(row[0]) == expense_id:
                    row[2] = description or row[2]
                    row[3] = amount or row[3]
                    row[4] = category or row[4]
                    updated = True
                rows.append(row)

        if updated:
            with open(self.csv_file, 'w', newline='') as f:
                writer = csv.writer(f)
                writer.writerows(rows)
        return updated

    def list_expenses(self):
        with open(self.csv_file, 'r') as f:
            reader = csv.reader(f)
            return [row for row in reader if row]

def demo():
    tracker = ExpenseTracker()

    # Add sample expenses
    id1 = tracker.add_expense("Groceries", 45.67, "Food")
    id2 = tracker.add_expense("Bus ticket", 2.5, "Transport")

    print("=== Expense Tracker Demo ===")
    print(f"Added expenses with IDs: {id1}, {id2}\n")

    # List current expenses
    print("Current expenses:")
    for expense in tracker.list_expenses():
        print(f"ID {expense[0]}: {expense[1]} - {expense[2]} (${expense[3]}) [{expense[4]}]")

    # Update an expense
    tracker.update_expense(id1, "Weekly groceries", 42.33)

    print("\nUpdated expense ID 1\n")
    print("Updated expenses:")
    for expense in tracker.list_expenses():
        print(f"ID {expense[0]}: {expense[1]} - {expense[2]} (${expense[3]}) [{expense[4]}]")

if __name__ == "__main__":
    demo()
```
