# Simple Expense Tracker in Python

## Overview
A lightweight expense tracker with CSV storage, supporting add, update, and list operations.

## Key Features
- **Storage**: CSV backend (`expenses.csv`)
- **Operations**: Add, update, list expenses
- **Demo**: Sample run included

## Code Implementation

```python
import csv
from datetime import datetime
from pathlib import Path

class ExpenseTracker:
    def __init__(self, csv_file="expenses.csv"):
        self.csv_file = Path(csv_file)
        self._init_csv()

    def _init_csv(self):
        if not self.csv_file.exists():
            with open(self.csv_file, 'w', newline='') as f:
                writer = csv.writer(f)
                writer.writerow(["id", "date", "description", "amount"])

    def add_expense(self, description, amount):
        expense_id = self._next_id()
        date = datetime.now().strftime("%Y-%m-%d")
        with open(self.csv_file, 'a', newline='') as f:
            writer = csv.writer(f)
            writer.writerow([expense_id, date, description, amount])
        return expense_id

    def update_expense(self, expense_id, new_description=None, new_amount=None):
        rows = []
        updated = False
        with open(self.csv_file, 'r') as f:
            reader = csv.reader(f)
            for row in reader:
                if row[0] == str(expense_id):
                    if new_description:
                        row[2] = new_description
                    if new_amount:
                        row[3] = new_amount
                    updated = True
                rows.append(row)

        if updated:
            with open(self.csv_file, 'w', newline='') as f:
                writer = csv.writer(f)
                writer.writerows(rows)
        return updated

    def list_expenses(self):
        with open(self.csv_file, 'r') as f:
            reader = csv.reader(f)
            return list(reader)[1:]  # Skip header

    def _next_id(self):
        with open(self.csv_file, 'r') as f:
            reader = csv.reader(f)
            rows = list(reader)
            return str(int(rows[-1][0]) + 1) if rows else "1"
```

## Demo Run

```python
# Initialize tracker
tracker = ExpenseTracker()

# Add expenses
id1 = tracker.add_expense("Groceries", 45.67)
id2 = tracker.add_expense("Coffee", 3.50)

# List all expenses
print("Initial expenses:")
for expense in tracker.list_expenses():
    print(f"ID: {expense[0]}, Date: {expense[1]}, Desc: {expense[2]}, Amount: ${expense[3]}")

# Update an expense
tracker.update_expense(id2, new_amount="4.00")

# List after update
print("\nAfter updating coffee price:")
for expense in tracker.list_expenses():
    print(f"ID: {expense[0]}, Date: {expense[1]}, Desc: {expense[2]}, Amount: ${expense[3]}")
```
