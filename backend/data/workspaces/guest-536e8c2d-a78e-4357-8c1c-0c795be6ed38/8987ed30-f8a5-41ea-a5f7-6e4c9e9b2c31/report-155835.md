# Build a simple expense tracker in Python with a CSV storage layer, add/update/list features and a demo run

# Simple Expense Tracker in Python

## Overview
A lightweight command-line expense tracker with CSV storage, supporting add, update, and list operations.

## Key Features
- **Add expenses**: Record date, description, category, and amount
- **Update expenses**: Modify existing entries by ID
- **List expenses**: View all expenses with filtering options
- **CSV storage**: Persistent data storage in `expenses.csv`

## Code Implementation

```python
import csv
import os
from datetime import datetime

CSV_FILE = "expenses.csv"

def initialize_csv():
    if not os.path.exists(CSV_FILE):
        with open(CSV_FILE, mode='w', newline='') as file:
            writer = csv.writer(file)
            writer.writerow(["id", "date", "description", "category", "amount"])

def add_expense(date, description, category, amount):
    with open(CSV_FILE, mode='a', newline='') as file:
        writer = csv.writer(file)
        expense_id = len(list(csv.reader(open(CSV_FILE))))  # Simple ID generation
        writer.writerow([expense_id, date, description, category, amount])

def update_expense(expense_id, date=None, description=None, category=None, amount=None):
    rows = []
    with open(CSV_FILE, mode='r') as file:
        reader = csv.reader(file)
        rows = list(reader)

    for i, row in enumerate(rows):
        if row[0] == expense_id:
            if date: rows[i][1] = date
            if description: rows[i][2] = description
            if category: rows[i][3] = category
            if amount: rows[i][4] = amount
            break

    with open(CSV_FILE, mode='w', newline='') as file:
        writer = csv.writer(file)
        writer.writerows(rows)

def list_expenses(category=None):
    with open(CSV_FILE, mode='r') as file:
        reader = csv.reader(file)
        next(reader)  # Skip header
        expenses = list(reader)

        if category:
            expenses = [e for e in expenses if e[3].lower() == category.lower()]

        for expense in expenses:
            print(f"ID: {expense[0]}, Date: {expense[1]}, Desc: {expense[2]}, "
                  f"Cat: {expense[3]}, Amt: ${expense[4]}")

def demo():
    initialize_csv()

    # Add sample expenses
    add_expense("2023-11-01", "Groceries", "Food", "50.25")
    add_expense("2023-11-02", "Gas", "Transport", "35.50")
    add_expense("2023-11-03", "Movie tickets", "Entertainment", "24.00")

    # List all expenses
    print("\nAll Expenses:")
    list_expenses()

    # List food expenses
    print("\nFood Expenses:")
    list_expenses("Food")

    # Update an expense
    update_expense("1", amount="55.00")

    print("\nAfter Update:")
    list_expenses()

if __name__ == "__main__":
    demo()
```
